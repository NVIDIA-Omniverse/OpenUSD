cmake_minimum_required(VERSION 3.26)
project(ModuleSplit LANGUAGES CXX)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake/defaults"
    "${CMAKE_CURRENT_SOURCE_DIR}/cmake/macros")
include(Public)
set(BUILD_SHARED_LIBS ON)
set(PXR_BUILD_TESTS OFF)
set(PXR_ENABLE_PRECOMPILED_HEADERS OFF)
set(PXR_BUILD_MONOLITHIC OFF)
set(PXR_INSTALL_SUBDIR "")
set(PXR_LIB_PREFIX "")
add_custom_target(shared_libs)
if(PXR_ENABLE_PYTHON_SUPPORT)
    find_package(pxr CONFIG REQUIRED COMPONENTS tfBoostPython
        PATHS "${USD_PREFIX}" NO_DEFAULT_PATH)
    set(pythonComponents Interpreter Development)
    if(WIN32)
        list(APPEND pythonComponents Development.SABIModule)
    endif()
    find_package(Python3 "${Python3_VERSION_MAJOR}.${Python3_VERSION_MINOR}" EXACT REQUIRED
        COMPONENTS ${pythonComponents})
    set(PYTHON_INCLUDE_DIRS "${Python3_INCLUDE_DIRS}")
    set(PYTHON_LIBRARIES Python3::Python)
    if(WIN32)
        set(Python3_LIBRARY "${Python3_LIBRARIES}" CACHE FILEPATH "Selected full Python library" FORCE)
    endif()
    set(PYTHON_EXECUTABLE "${Python3_EXECUTABLE}")
    set(PXR_PYTHON_SHEBANG "${PYTHON_EXECUTABLE}")
    add_custom_target(python_modules ALL)
endif()
if(OBSOLETE_ARGUMENTS)
    pxr_library(obsolete PYMODULE_CPPFILES nonexistent.cpp)
endif()
add_subdirectory(native)
if(TARGET _nativeProbe OR TARGET _bindingProbe)
    message(FATAL_ERROR "pxr_library created an implicit Python module")
endif()
get_target_property(before nativeProbe SOURCES)
if(before MATCHES "moduleDeps|module[.]cpp")
    message(FATAL_ERROR "Native library acquired bindings before their declaration")
endif()
if(VERIFY_MONOLITHIC_LINKS AND PXR_ENABLE_PYTHON_SUPPORT)
    # Model a core requirement while using the installed non-monolithic SDK.
    # This tests the logical target mapping without needing a second SDK build.
    add_library(coreDependency INTERFACE)
    add_library(usd_m INTERFACE)
    target_link_libraries(usd_m INTERFACE tf)
    set(PXR_CORE_LIBS coreDependency)
    set(PXR_BUILD_MONOLITHIC ON)
endif()
add_subdirectory(bindings)
add_subdirectory(pure)
add_subdirectory(resources)
if(TARGET resourceProbe OR TARGET _resourceProbe)
    message(FATAL_ERROR "Resource-only package acquired a compiled target")
endif()
get_target_property(after nativeProbe SOURCES)
get_target_property(links nativeProbe LINK_LIBRARIES)
if(links MATCHES "bindingOnly|BoostPython")
    message(FATAL_ERROR "Module dependencies entered the native link interface")
endif()
if(PXR_ENABLE_PYTHON_SUPPORT)
    if(NOT TARGET _bindingProbe OR TARGET _pureProbe OR NOT after MATCHES "moduleDeps[.]cpp")
        message(FATAL_ERROR "Explicit compiled/pure module ownership is incorrect")
    endif()
    get_target_property(moduleLinks _bindingProbe LINK_LIBRARIES)
    if(NOT "nativeProbe" IN_LIST moduleLinks OR NOT "bindingOnly" IN_LIST moduleLinks)
        message(FATAL_ERROR "Module lost its native or binding-only requirements")
    endif()
    if(VERIFY_MONOLITHIC_LINKS)
        if(NOT "usd_m" IN_LIST moduleLinks OR "coreDependency" IN_LIST moduleLinks)
            message(FATAL_ERROR "Module-only core dependencies bypassed the monolith")
        endif()
        set(BUILD_SHARED_LIBS OFF)
        _pxr_resolve_link_libraries(_bindingProbe staticLinks coreDependency)
        if(NOT "$<LINK_LIBRARY:WHOLE_ARCHIVE,usd_m>" IN_LIST staticLinks)
            message(FATAL_ERROR "Static monolithic module dependency lost whole-archive linkage")
        endif()
        set(BUILD_SHARED_LIBS ON)
    endif()
else()
    if(TARGET _bindingProbe OR after MATCHES "moduleDeps[.]cpp")
        message(FATAL_ERROR "Python-disabled build acquired module implementation")
    endif()
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/native-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:nativeProbe>")
if(PXR_ENABLE_PYTHON_SUPPORT)
    install(FILES pxr-init.py DESTINATION lib/python/pxr RENAME __init__.py)
endif()
