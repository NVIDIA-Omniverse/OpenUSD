cmake_minimum_required(VERSION 3.26)
project(UnixPythonRuntime LANGUAGES CXX)
list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}"
    "${USD_SOURCE}/cmake/defaults" "${USD_SOURCE}/cmake/macros")
include(Private)

# Exercise both Unix branches on any build host; no binaries are built here.
set(WIN32 FALSE)
set(APPLE "${SIMULATE_APPLE}")
if(PACKAGE_COMPONENT)
    set(PXR_ENABLE_PYTHON_SUPPORT ON)
    set(PXR_BUILD_MONOLITHIC OFF)
    set(BUILD_SHARED_LIBS ON)
    set(PXR_ALL_LIBS tf)
    set(Python3_VERSION_MAJOR 3)
    set(Python3_VERSION_MINOR 12)
    set(PXR_FIND_TBB_IN_CONFIG OFF)
    set(PXR_ENABLE_MATERIALX_SUPPORT OFF)
    set(Imath_FOUND OFF)
    set(prefix "${CMAKE_CURRENT_BINARY_DIR}/sdk")
    file(MAKE_DIRECTORY "${prefix}/cmake")
    configure_file("${USD_SOURCE}/pxr/pxrConfig.cmake.in" "${prefix}/pxrConfig.cmake" @ONLY)
    file(WRITE "${prefix}/cmake/pxrTargets.cmake" "
add_library(pythonRuntime INTERFACE IMPORTED)
set_property(TARGET pythonRuntime PROPERTY INTERFACE_LINK_LIBRARIES Python3::SABIModule)
add_library(tf INTERFACE IMPORTED)
set_property(TARGET tf PROPERTY INTERFACE_LINK_LIBRARIES pythonRuntime)
add_library(tfBoostPython INTERFACE IMPORTED)
set_property(TARGET tfBoostPython PROPERTY INTERFACE_LINK_LIBRARIES \"tf;Python3::Python\")
")
    if(PACKAGE_COMPONENT STREQUAL "default")
        find_package(pxr CONFIG REQUIRED PATHS "${prefix}" NO_DEFAULT_PATH)
    else()
        find_package(pxr CONFIG REQUIRED COMPONENTS ${PACKAGE_COMPONENT}
            PATHS "${prefix}" NO_DEFAULT_PATH)
    endif()
    if(PACKAGE_COMPONENT STREQUAL "tf")
        get_property(exact GLOBAL PROPERTY pythonRequestExact)
        get_property(version GLOBAL PROPERTY pythonRequestVersion)
        if(TARGET Python3::Python OR NOT TARGET Python3::SABIModule
                OR exact OR NOT version STREQUAL "3.10")
            message(FATAL_ERROR "Core discovery selected the wrong Python contract")
        endif()
    endif()
    return()
endif()

file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/empty.cpp" "int main() { return 0; }\n")
add_library(native SHARED "${CMAKE_CURRENT_BINARY_DIR}/empty.cpp")
add_library(pythonFree INTERFACE)
add_library(proxy INTERFACE)
target_link_libraries(proxy INTERFACE native)
add_executable(embedding "${CMAKE_CURRENT_BINARY_DIR}/empty.cpp")
target_link_libraries(embedding PRIVATE proxy)
add_executable(freeApplication "${CMAKE_CURRENT_BINARY_DIR}/empty.cpp")
target_link_libraries(freeApplication PRIVATE pythonFree)
add_library(monolith STATIC "${CMAKE_CURRENT_BINARY_DIR}/empty.cpp")
target_link_libraries(monolith PUBLIC native)
add_executable(monolithicApplication "${CMAKE_CURRENT_BINARY_DIR}/empty.cpp")
target_link_libraries(monolithicApplication PRIVATE "$<LINK_LIBRARY:WHOLE_ARCHIVE,monolith>")
if(PXR_ENABLE_PYTHON_SUPPORT)
    # The old full runtime variable must not enter native dependencies.
    find_package(Python3 REQUIRED COMPONENTS Development.Embed)
    set(PYTHON_LIBRARIES Python3::Python)
    _pxr_setup_python_runtime()
    target_link_libraries(native PUBLIC pythonRuntime)
    get_target_property(links pythonRuntime INTERFACE_LINK_LIBRARIES)
    if(NOT links STREQUAL "Python3::SABIModule")
        message(FATAL_ERROR "Native runtime retains versioned linkage: ${links}")
    endif()
    if(INVALID_NATIVE_DEPENDENCY)
        target_link_libraries(native PRIVATE Python3::Python)
    endif()
endif()
_pxr_apply_native_python_policy(native)
_pxr_link_python_executables("${CMAKE_CURRENT_SOURCE_DIR}")
_pxr_validate_native_python_targets("${CMAKE_CURRENT_SOURCE_DIR}")
get_target_property(links embedding LINK_LIBRARIES)
if(PXR_ENABLE_PYTHON_SUPPORT AND NOT "Python3::Python" IN_LIST links)
    message(FATAL_ERROR "Embedding application lost its interpreter dependency")
elseif(NOT PXR_ENABLE_PYTHON_SUPPORT AND TARGET pythonRuntime)
    message(FATAL_ERROR "Python-disabled build acquired a runtime")
endif()
get_target_property(links freeApplication LINK_LIBRARIES)
if("Python3::Python" IN_LIST links)
    message(FATAL_ERROR "Python-free application acquired an interpreter dependency")
endif()
get_target_property(links monolithicApplication LINK_LIBRARIES)
if(PXR_ENABLE_PYTHON_SUPPORT AND NOT "Python3::Python" IN_LIST links)
    message(FATAL_ERROR "Static monolithic application lost its interpreter dependency")
endif()
