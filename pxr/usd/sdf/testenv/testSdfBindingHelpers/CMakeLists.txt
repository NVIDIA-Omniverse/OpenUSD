cmake_minimum_required(VERSION 3.26)
project(SdfBindingHelpers LANGUAGES CXX)
file(TO_CMAKE_PATH "${USD_PREFIX}" USD_PREFIX)
find_package(pxr CONFIG REQUIRED COMPONENTS sdfBoostPython PATHS "${USD_PREFIX}" NO_DEFAULT_PATH)
get_target_property(kind sdfBoostPython TYPE)
if(NOT kind STREQUAL "INTERFACE_LIBRARY")
    message(FATAL_ERROR "Sdf binding helpers must not add a compiled library")
endif()
get_target_property(dependencies sdfBoostPython INTERFACE_LINK_LIBRARIES)
if(NOT "tfBoostPython" IN_LIST dependencies)
    message(FATAL_ERROR "Missing Boost binding requirements")
endif()
set(core sdf)
if(TARGET usd_m)
    set(core usd_m)
    if("sdf" IN_LIST dependencies)
        message(FATAL_ERROR "Monolithic helpers must select core through tfBoostPython")
    endif()
elseif(NOT "sdf" IN_LIST dependencies)
    message(FATAL_ERROR "Ordinary helpers must supply core Sdf")
endif()
get_target_property(headers sdfBoostPython HEADER_SET)
list(LENGTH headers count)
if(NOT count EQUAL 8)
    message(FATAL_ERROR "Expected eight binding helper headers, found ${count}")
endif()
set(headerSource "#include \"pxr/base/tf/boost/pySignatureExt.h\"\n")
foreach(name pyChildrenProxy pyChildrenView pyListEditorProxy pyListOp pyListProxy pyMapEditProxy pySpec pyUtils)
    set(path "${USD_PREFIX}/include/pxr/usd/sdf/boost/${name}.h")
    if(NOT path IN_LIST headers)
        message(FATAL_ERROR "Binding interface does not own ${name}")
    endif()
    string(APPEND headerSource "#include \"pxr/usd/sdf/boost/${name}.h\"\n")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/headers.cpp" "${headerSource}")
add_library(bindingHeaders OBJECT "${CMAKE_CURRENT_BINARY_DIR}/headers.cpp")
target_link_libraries(bindingHeaders PRIVATE sdfBoostPython)
target_compile_definitions(bindingHeaders PRIVATE MFB_PACKAGE_MODULE=HeaderConsumer)
add_library(sdfBindingHelpers MODULE module.cpp)
target_link_libraries(sdfBindingHelpers PRIVATE sdfBoostPython)
set_target_properties(sdfBindingHelpers PROPERTIES PREFIX "")
if(WIN32)
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
    set_target_properties(sdfBindingHelpers PROPERTIES SUFFIX ".pyd")
elseif(APPLE)
    target_link_options(sdfBindingHelpers PRIVATE "LINKER:-undefined,dynamic_lookup")
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/module-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:sdfBindingHelpers>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/core-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:${core}>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/runtime-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE_DIR:TBB::tbb>")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/linker.txt" "${CMAKE_LINKER}")
