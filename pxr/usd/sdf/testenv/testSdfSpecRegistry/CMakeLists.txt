cmake_minimum_required(VERSION 3.26)
project(SdfSpecRegistry LANGUAGES CXX)
find_package(pxr CONFIG REQUIRED COMPONENTS sdfBoostPython PATHS "${USD_PREFIX}" NO_DEFAULT_PATH)
set(core sdf)
if(TARGET usd_m)
    set(core usd_m)
endif()
if(WIN32)
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
add_library(holderNative SHARED native.cpp)
target_compile_definitions(holderNative PRIVATE HOLDER_NATIVE_EXPORTS)
target_compile_features(holderNative PUBLIC cxx_std_17)
target_link_libraries(holderNative PUBLIC ${core})
add_executable(holderContracts contracts.cpp)
target_link_libraries(holderContracts PRIVATE holderNative)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/header.cpp"
    "#include \"pxr/usd/sdf/pySpecRegistry.h\"\n#if defined(EXPECT_FULL_API) && defined(Py_LIMITED_API)\n#error Limited API leaks into consumers\n#endif\n")
foreach(mode Full Limited)
    add_library(registryHeader${mode} OBJECT "${CMAKE_CURRENT_BINARY_DIR}/header.cpp")
    target_link_libraries(registryHeader${mode} PRIVATE ${core})
    target_compile_features(registryHeader${mode} PRIVATE cxx_std_17)
endforeach()
target_compile_definitions(registryHeaderFull PRIVATE EXPECT_FULL_API)
add_library(registrySourceLimited OBJECT ../../pySpecRegistry.cpp)
target_link_libraries(registrySourceLimited PRIVATE ${core})
target_compile_features(registrySourceLimited PRIVATE cxx_std_17)
target_compile_definitions(registrySourceLimited PRIVATE SDF_EXPORTS
    MFB_PACKAGE_NAME=sdf MFB_ALT_PACKAGE_NAME=sdf MFB_PACKAGE_MODULE=Sdf)
foreach(module sdfSpecRegister sdfSpecConsume)
    if(module STREQUAL "sdfSpecRegister")
        set(source register.cpp)
    else()
        set(source consume.cpp)
    endif()
    add_library(${module} MODULE ${source})
    target_link_libraries(${module} PRIVATE holderNative)
    if(WIN32)
        target_link_libraries(${module} PRIVATE Python3::SABIModule)
    endif()
endforeach()
set(nativeTargets holderNative holderContracts registryHeaderLimited registrySourceLimited
    sdfSpecRegister sdfSpecConsume)
foreach(target IN LISTS nativeTargets)
    target_compile_definitions(${target} PRIVATE Py_LIMITED_API=0x030A0000 Py_NO_LINK_LIB)
endforeach()
foreach(target IN LISTS nativeTargets ITEMS registryHeaderFull)
    if(MSVC)
        target_compile_options(${target} PRIVATE /showIncludes)
    else()
        target_compile_options(${target} PRIVATE -H)
    endif()
endforeach()
add_library(sdfSpecBindingConsumer MODULE binding.cpp)
target_link_libraries(sdfSpecBindingConsumer PRIVATE holderNative sdfBoostPython)
target_compile_definitions(sdfSpecBindingConsumer PRIVATE MFB_PACKAGE_MODULE=DownstreamConsumer)
foreach(module sdfSpecRegister sdfSpecConsume sdfSpecBindingConsumer)
    set_target_properties(${module} PROPERTIES PREFIX "")
    if(WIN32)
        set_target_properties(${module} PROPERTIES SUFFIX ".pyd")
    elseif(APPLE)
        target_link_options(${module} PRIVATE "LINKER:-undefined,dynamic_lookup")
    endif()
endforeach()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/native-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:holderNative>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/contracts-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:holderContracts>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/module-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:sdfSpecConsume>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/runtime-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE_DIR:TBB::tbb>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/core-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:${core}>")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/linker.txt" "${CMAKE_LINKER}")
