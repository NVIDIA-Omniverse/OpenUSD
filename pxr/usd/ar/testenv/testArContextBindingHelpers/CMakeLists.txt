cmake_minimum_required(VERSION 3.26)
project(ArContextBindingHelpers LANGUAGES CXX)
file(TO_CMAKE_PATH "${USD_PREFIX}" USD_PREFIX)
find_package(pxr CONFIG REQUIRED COMPONENTS arBoostPython
    PATHS "${USD_PREFIX}" NO_DEFAULT_PATH)
if(WIN32)
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
set(core ar)
if(TARGET usd_m)
    set(core usd_m)
    get_target_property(coreType usd_m TYPE)
    if(coreType STREQUAL "STATIC_LIBRARY")
        set(core "$<LINK_LIBRARY:WHOLE_ARCHIVE,usd_m>")
    endif()
endif()
add_library(arContextNative SHARED native.cpp)
target_compile_definitions(arContextNative PRIVATE AR_CONTEXT_NATIVE_EXPORTS)
target_link_libraries(arContextNative PUBLIC ${core})
target_compile_features(arContextNative PUBLIC cxx_std_17)
add_executable(contextContracts contracts.cpp)
target_link_libraries(contextContracts PRIVATE arContextNative)
foreach(target arContextNative contextContracts)
    target_compile_definitions(${target} PRIVATE Py_LIMITED_API=0x030A0000)
    if(MSVC)
        target_compile_options(${target} PRIVATE /showIncludes)
    endif()
endforeach()
foreach(mode full limited)
    add_library(contextHeader_${mode} OBJECT native.cpp)
    target_compile_definitions(contextHeader_${mode} PRIVATE AR_CONTEXT_NATIVE_EXPORTS)
    target_link_libraries(contextHeader_${mode} PRIVATE ${core})
    target_compile_features(contextHeader_${mode} PRIVATE cxx_std_17)
    if(mode STREQUAL "limited")
        target_compile_definitions(contextHeader_${mode} PRIVATE Py_LIMITED_API=0x030A0000)
    endif()
    if(MSVC)
        target_compile_options(contextHeader_${mode} PRIVATE /showIncludes)
    endif()
endforeach()
add_library(contextSourceLimited OBJECT ../../resolverContext.cpp)
target_link_libraries(contextSourceLimited PRIVATE ${core})
target_compile_features(contextSourceLimited PRIVATE cxx_std_17)
target_compile_definitions(contextSourceLimited PRIVATE
    AR_EXPORTS Py_LIMITED_API=0x030A0000)
if(MSVC)
    target_compile_options(contextSourceLimited PRIVATE /showIncludes)
endif()
foreach(module arContextConsume arContextRegister)
    if(module STREQUAL "arContextConsume")
        set(source consume.cpp)
    else()
        set(source register.cpp)
    endif()
    add_library(${module} MODULE ${source})
    target_link_libraries(${module} PRIVATE arBoostPython arContextNative)
    set_target_properties(${module} PROPERTIES PREFIX "")
    if(WIN32)
        set_target_properties(${module} PROPERTIES SUFFIX ".pyd")
    elseif(APPLE)
        target_link_options(${module} PRIVATE "LINKER:-undefined,dynamic_lookup")
    endif()
endforeach()
get_target_property(bindingHeaders arBoostPython HEADER_SET)
if(NOT "${USD_PREFIX}/include/pxr/usd/ar/boost/pyResolverContext.h" IN_LIST bindingHeaders)
    message(FATAL_ERROR "arBoostPython does not own its binding header")
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/contracts-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:contextContracts>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/native-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:arContextNative>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/module-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:arContextConsume>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/runtime-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE_DIR:TBB::tbb>")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/linker.txt" "${CMAKE_LINKER}")
get_target_property(bindingRuntimeType python TYPE)
if(bindingRuntimeType STREQUAL "SHARED_LIBRARY")
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/binding-$<CONFIG>.txt"
        CONTENT "$<TARGET_FILE_NAME:python>")
else()
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/binding-$<CONFIG>.txt"
        CONTENT "")
endif()
