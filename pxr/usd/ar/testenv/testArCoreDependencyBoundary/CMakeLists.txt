cmake_minimum_required(VERSION 3.26)
project(ArCoreDependencyBoundary LANGUAGES CXX)

find_package(pxr CONFIG REQUIRED COMPONENTS ar PATHS "${USD_PREFIX}" NO_DEFAULT_PATH)
if(NOT PXR_LIBRARIES STREQUAL "ar")
    message(FATAL_ERROR "Ar core component exposes unrelated libraries: ${PXR_LIBRARIES}")
endif()
if(TARGET Python3::Python)
    message(FATAL_ERROR "Core Ar requested version-specific Python")
endif()
get_target_property(features ar INTERFACE_COMPILE_FEATURES)
if(NOT "cxx_std_17" IN_LIST features)
    message(FATAL_ERROR "Core Ar must supply its C++17 requirement")
endif()
get_target_property(definitions ar INTERFACE_COMPILE_DEFINITIONS)
if(definitions MATCHES "Py_LIMITED_API")
    message(FATAL_ERROR "The core Limited API definition leaks to bindings")
endif()

function(check_dependencies target)
    get_property(visited GLOBAL PROPERTY checkedTargets)
    if(target IN_LIST visited)
        return()
    endif()
    set_property(GLOBAL APPEND PROPERTY checkedTargets "${target}")
    get_target_property(dependencies "${target}" INTERFACE_LINK_LIBRARIES)
    foreach(dependency IN LISTS dependencies)
        if(dependency MATCHES "(^|::)(boost|python)(::|$)|(tf|vt|ar)BoostPython"
                OR dependency STREQUAL "Python3::Python")
            message(FATAL_ERROR "${target} exposes binding dependency ${dependency}")
        endif()
        if(TARGET "${dependency}")
            check_dependencies("${dependency}")
        endif()
    endforeach()
endfunction()
check_dependencies(ar)

if(WIN32)
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
endif()
file(STRINGS "${CORE_HEADERS}" headers)
set(headerSource "")
foreach(header IN LISTS headers)
    string(APPEND headerSource "#include \"pxr/usd/ar/${header}\"\n")
endforeach()
string(APPEND headerSource "
#if defined(AR_EXPECT_FULL_API) && defined(Py_LIMITED_API)
#error The core must not impose the Limited API on its consumers.
#endif
")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/headers.cpp" "${headerSource}")
foreach(mode Full Limited)
    add_library(arHeaders${mode} OBJECT "${CMAKE_CURRENT_BINARY_DIR}/headers.cpp")
    target_link_libraries(arHeaders${mode} PRIVATE ar)
    if(mode STREQUAL "Limited")
        target_compile_definitions(arHeaders${mode} PRIVATE Py_LIMITED_API=0x030A0000)
    else()
        target_compile_definitions(arHeaders${mode} PRIVATE AR_EXPECT_FULL_API)
    endif()
endforeach()

file(STRINGS "${CORE_SOURCES}" sources)
add_library(arSourcesLimited OBJECT ${sources})
target_link_libraries(arSourcesLimited PRIVATE ar)
# Implementation units also need Ar's private headers, which are intentionally
# absent from the installed SDK. Keep installed-header checks above independent.
get_filename_component(openusdSource "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
target_include_directories(arSourcesLimited PRIVATE "${openusdSource}")
target_compile_definitions(arSourcesLimited PRIVATE AR_EXPORTS
    Py_LIMITED_API=0x030A0000 MFB_PACKAGE_NAME=ar MFB_ALT_PACKAGE_NAME=ar MFB_PACKAGE_MODULE=Ar)
foreach(target arHeadersFull arHeadersLimited arSourcesLimited)
    if(MSVC)
        target_compile_options(${target} PRIVATE /showIncludes)
    else()
        target_compile_options(${target} PRIVATE -H)
    endif()
endforeach()

add_executable(arCoreConsumer consumer.cpp)
target_link_libraries(arCoreConsumer PRIVATE ar)
if(NOT WIN32)
    find_package(Python3 3.10 REQUIRED COMPONENTS Development.Embed)
    target_link_libraries(arCoreConsumer PRIVATE Python3::Python)
endif()
target_compile_definitions(arCoreConsumer PRIVATE Py_LIMITED_API=0x030A0000)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/consumer-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:arCoreConsumer>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/core-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:ar>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/runtime-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE_DIR:TBB::tbb>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/binding-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE_NAME:python>")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/linker.txt" "${CMAKE_LINKER}")
