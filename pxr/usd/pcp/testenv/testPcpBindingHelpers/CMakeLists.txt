cmake_minimum_required(VERSION 3.26)
project(PcpBindingHelpers LANGUAGES CXX)
file(TO_CMAKE_PATH "${USD_PREFIX}" USD_PREFIX)
find_package(pxr CONFIG REQUIRED COMPONENTS pcpBoostPython PATHS "${USD_PREFIX}" NO_DEFAULT_PATH)
get_target_property(kind pcpBoostPython TYPE)
if(NOT kind STREQUAL "INTERFACE_LIBRARY")
    message(FATAL_ERROR "Pcp binding helpers must not add a compiled library")
endif()
get_target_property(dependencies pcpBoostPython INTERFACE_LINK_LIBRARIES)
if(NOT "tfBoostPython" IN_LIST dependencies)
    message(FATAL_ERROR "Missing Boost binding requirements")
endif()
set(core pcp)
if(TARGET usd_m)
    set(core usd_m)
    if("pcp" IN_LIST dependencies)
        message(FATAL_ERROR "Monolithic helpers must select core through tfBoostPython")
    endif()
elseif(NOT "pcp" IN_LIST dependencies)
    message(FATAL_ERROR "Ordinary helpers must supply core Pcp")
endif()
get_target_property(headers pcpBoostPython HEADER_SET)
list(LENGTH headers count)
if(NOT count EQUAL 1 OR NOT "${USD_PREFIX}/include/pxr/usd/pcp/boost/pyUtils.h" IN_LIST headers)
    message(FATAL_ERROR "Binding interface must own exactly the Pcp adapter header")
endif()
# Compile the installed header on its own, without a prefix header.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/header.cpp"
    "#include \"pxr/usd/pcp/boost/pyUtils.h\"\n")
add_library(bindingHeader OBJECT "${CMAKE_CURRENT_BINARY_DIR}/header.cpp")
target_link_libraries(bindingHeader PRIVATE pcpBoostPython)
add_library(pcpBindingHelpers MODULE module.cpp)
target_link_libraries(pcpBindingHelpers PRIVATE pcpBoostPython)
foreach(target bindingHeader pcpBindingHelpers)
    target_compile_features(${target} PRIVATE cxx_std_17)
endforeach()
set_target_properties(pcpBindingHelpers PROPERTIES PREFIX "")
if(WIN32)
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
    set_target_properties(pcpBindingHelpers PROPERTIES SUFFIX ".pyd")
elseif(APPLE)
    target_link_options(pcpBindingHelpers PRIVATE "LINKER:-undefined,dynamic_lookup")
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/module-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:pcpBindingHelpers>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/core-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:${core}>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/runtime-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE_DIR:TBB::tbb>")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/linker.txt" "${CMAKE_LINKER}")
