cmake_minimum_required(VERSION 3.26)
project(PcpCoreDependencyBoundary LANGUAGES CXX)

find_package(pxr CONFIG REQUIRED COMPONENTS pcp PATHS "${USD_PREFIX}" NO_DEFAULT_PATH)
if(NOT PXR_LIBRARIES STREQUAL "pcp")
    message(FATAL_ERROR "Pcp core component exposes unrelated libraries: ${PXR_LIBRARIES}")
endif()
if(TARGET Python3::Python)
    message(FATAL_ERROR "Core Pcp requested version-specific Python")
endif()
get_target_property(features pcp INTERFACE_COMPILE_FEATURES)
if(NOT "cxx_std_17" IN_LIST features)
    message(FATAL_ERROR "Core Pcp must supply its C++17 requirement")
endif()
get_target_property(definitions pcp INTERFACE_COMPILE_DEFINITIONS)
if(definitions MATCHES "Py_LIMITED_API")
    message(FATAL_ERROR "The core Limited API definition leaks to bindings")
endif()

function(check_dependencies target)
    get_property(visited GLOBAL PROPERTY checkedTargets)
    if(target IN_LIST visited)
        return()
    endif()
    set_property(GLOBAL APPEND PROPERTY checkedTargets "${target}")
    get_target_property(dependencies "${target}" INTERFACE_LINK_LIBRARIES)
    foreach(dependency IN LISTS dependencies)
        if(dependency MATCHES "(^|::)(boost|python)(::|$)|(tf|vt|ar|sdf|pcp)BoostPython"
                OR dependency STREQUAL "Python3::Python")
            message(FATAL_ERROR "${target} exposes binding dependency ${dependency}")
        endif()
        if(TARGET "${dependency}")
            check_dependencies("${dependency}")
        endif()
    endforeach()
endfunction()
check_dependencies(pcp)

if(WIN32)
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
endif()
file(STRINGS "${CORE_HEADERS}" headers)
set(headerSource "")
foreach(header IN LISTS headers)
    string(APPEND headerSource "#include \"pxr/usd/pcp/${header}\"\n")
endforeach()
string(APPEND headerSource "
#if defined(PCP_EXPECT_FULL_API) && defined(Py_LIMITED_API)
#error The core must not impose the Limited API on its consumers.
#endif
")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/headers.cpp" "${headerSource}")
foreach(mode Full Limited)
    add_library(pcpHeaders${mode} OBJECT "${CMAKE_CURRENT_BINARY_DIR}/headers.cpp")
    target_link_libraries(pcpHeaders${mode} PRIVATE pcp)
    if(mode STREQUAL "Limited")
        target_compile_definitions(pcpHeaders${mode} PRIVATE Py_LIMITED_API=0x030A0000)
    else()
        target_compile_definitions(pcpHeaders${mode} PRIVATE PCP_EXPECT_FULL_API)
    endif()
endforeach()

file(STRINGS "${CORE_SOURCES}" sources)
# Implementation units also need Pcp's private headers, which are intentionally
# absent from the installed SDK. Keep installed-header checks above independent.
get_filename_component(openusdSource "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
# MSVC cannot combine /showIncludes with /MP. Separate object targets allow
# the build tool's --parallel limit to apply to this complete source audit.
set(sourceTargets "")
foreach(group RANGE 0 3)
    set(groupSources "")
    set(index 0)
    foreach(source IN LISTS sources)
        math(EXPR sourceGroup "${index} % 4")
        if(sourceGroup EQUAL group)
            list(APPEND groupSources "${source}")
        endif()
        math(EXPR index "${index} + 1")
    endforeach()
    set(target pcpSourcesLimited${group})
    add_library(${target} OBJECT ${groupSources})
    target_link_libraries(${target} PRIVATE pcp)
    target_include_directories(${target} PRIVATE "${openusdSource}")
    target_compile_definitions(${target} PRIVATE PCP_EXPORTS
        Py_LIMITED_API=0x030A0000 MFB_PACKAGE_NAME=pcp MFB_ALT_PACKAGE_NAME=pcp MFB_PACKAGE_MODULE=Pcp)
    list(APPEND sourceTargets ${target})
endforeach()
foreach(target pcpHeadersFull pcpHeadersLimited ${sourceTargets})
    if(MSVC)
        target_compile_options(${target} PRIVATE /showIncludes /bigobj)
    else()
        target_compile_options(${target} PRIVATE -H)
    endif()
endforeach()

add_executable(pcpCoreConsumer consumer.cpp)
target_link_libraries(pcpCoreConsumer PRIVATE pcp)
if(NOT WIN32)
    find_package(Python3 3.10 REQUIRED COMPONENTS Development.Embed)
    target_link_libraries(pcpCoreConsumer PRIVATE Python3::Python)
endif()
target_compile_definitions(pcpCoreConsumer PRIVATE Py_LIMITED_API=0x030A0000)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/consumer-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:pcpCoreConsumer>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/core-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:pcp>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/runtime-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE_DIR:TBB::tbb>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/binding-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE_NAME:python>")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/linker.txt" "${CMAKE_LINKER}")
