cmake_minimum_required(VERSION 3.26)
project(UsdBindingHelpers LANGUAGES CXX)
file(TO_CMAKE_PATH "${USD_PREFIX}" USD_PREFIX)
find_package(pxr CONFIG REQUIRED COMPONENTS usdBoostPython PATHS "${USD_PREFIX}" NO_DEFAULT_PATH)
get_target_property(kind usdBoostPython TYPE)
if(NOT kind STREQUAL "INTERFACE_LIBRARY")
    message(FATAL_ERROR "Usd binding helpers must not add a compiled library")
endif()
get_target_property(dependencies usdBoostPython INTERFACE_LINK_LIBRARIES)
if(NOT "tfBoostPython" IN_LIST dependencies)
    message(FATAL_ERROR "Missing Tf binding requirements")
endif()
set(core usd)
if(TARGET usd_m)
    set(core usd_m)
    if("usd" IN_LIST dependencies)
        message(FATAL_ERROR "Monolithic binding interface must select core through tfBoostPython")
    endif()
elseif(NOT "usd" IN_LIST dependencies)
    message(FATAL_ERROR "Ordinary binding interface must supply core Usd")
endif()
get_target_property(headers usdBoostPython HEADER_SET)
list(LENGTH headers count)
if(NOT count EQUAL 2)
    message(FATAL_ERROR "Expected exactly two owned binding headers")
endif()
foreach(name pyConversions pyEditContext)
    if(NOT "${USD_PREFIX}/include/pxr/usd/usd/boost/${name}.h" IN_LIST headers)
        message(FATAL_ERROR "Binding interface does not own ${name}")
    endif()
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${name}.cpp"
        "#include \"pxr/usd/usd/boost/${name}.h\"\n")
    add_library(${name}Header OBJECT "${CMAKE_CURRENT_BINARY_DIR}/${name}.cpp")
    target_link_libraries(${name}Header PRIVATE usdBoostPython)
    target_compile_features(${name}Header PRIVATE cxx_std_17)
endforeach()
# The carrier itself has no backend dependency, even though bindings own it.
add_library(contextHeaderLimited OBJECT "${CMAKE_CURRENT_BINARY_DIR}/pyEditContext.cpp")
target_link_libraries(contextHeaderLimited PRIVATE ${core})
target_compile_features(contextHeaderLimited PRIVATE cxx_std_17)
target_compile_definitions(contextHeaderLimited PRIVATE Py_LIMITED_API=0x030A0000)
if(MSVC)
    target_compile_options(contextHeaderLimited PRIVATE /showIncludes)
else()
    target_compile_options(contextHeaderLimited PRIVATE -H)
endif()
get_property(imported DIRECTORY PROPERTY IMPORTED_TARGETS)
foreach(target IN LISTS imported)
    if(NOT target MATCHES "BoostPython")
        get_target_property(links "${target}" INTERFACE_LINK_LIBRARIES)
        if("usdBoostPython" IN_LIST links)
            message(FATAL_ERROR "Core target ${target} exposes the binding interface")
        endif()
    endif()
endforeach()
add_library(usdBindingHelpers MODULE module.cpp)
target_link_libraries(usdBindingHelpers PRIVATE usdBoostPython)
target_compile_features(usdBindingHelpers PRIVATE cxx_std_17)
set_target_properties(usdBindingHelpers PROPERTIES PREFIX "")
if(WIN32)
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
    set_target_properties(usdBindingHelpers PROPERTIES SUFFIX ".pyd")
elseif(APPLE)
    target_link_options(usdBindingHelpers PRIVATE "LINKER:-undefined,dynamic_lookup")
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/module-$<CONFIG>.txt" CONTENT "$<TARGET_FILE:usdBindingHelpers>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/core-$<CONFIG>.txt" CONTENT "$<TARGET_FILE:${core}>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/runtime-$<CONFIG>.txt" CONTENT "$<TARGET_FILE_DIR:TBB::tbb>")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/linker.txt" "${CMAKE_LINKER}")
