cmake_minimum_required(VERSION 3.26)
project(VtArrayBindingHelpers LANGUAGES CXX)
file(TO_CMAKE_PATH "${USD_PREFIX}" USD_PREFIX)
find_package(pxr CONFIG REQUIRED COMPONENTS vtBoostPython
    PATHS "${USD_PREFIX}" NO_DEFAULT_PATH)
get_target_property(headers vtBoostPython HEADER_SET)
foreach(header wrapArray.h wrapArrayEdit.h pyOperators.h)
    if(NOT "${USD_PREFIX}/include/pxr/base/vt/boost/${header}" IN_LIST headers)
        message(FATAL_ERROR "Binding target does not own ${header}")
    endif()
endforeach()
if(WIN32)
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
add_library(vtArrayHelpers MODULE module.cpp extra.cpp)
target_link_libraries(vtArrayHelpers PRIVATE vtBoostPython)
target_compile_definitions(vtArrayHelpers PRIVATE MFB_PACKAGE_MODULE=vtArrayHelpers)
set_target_properties(vtArrayHelpers PROPERTIES PREFIX "")
if(WIN32)
    set_target_properties(vtArrayHelpers PROPERTIES SUFFIX ".pyd")
elseif(APPLE)
    target_link_options(vtArrayHelpers PRIVATE "LINKER:-undefined,dynamic_lookup")
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/module-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:vtArrayHelpers>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/runtime-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE_DIR:TBB::tbb>")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/linker.txt" "${CMAKE_LINKER}")
if(TARGET usd_m)
    set(bindingRuntime "")
else()
    set(bindingRuntime "$<TARGET_FILE_NAME:python>")
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/binding-$<CONFIG>.txt"
    CONTENT "${bindingRuntime}")
