cmake_minimum_required(VERSION 3.26)
project(VtCoreDependencyBoundary LANGUAGES CXX)

find_package(pxr CONFIG REQUIRED COMPONENTS vt PATHS "${USD_PREFIX}" NO_DEFAULT_PATH)
if(NOT PXR_LIBRARIES STREQUAL "vt")
    message(FATAL_ERROR "Vt core component exposes unrelated libraries: ${PXR_LIBRARIES}")
endif()
if(WIN32 AND TARGET Python3::Python)
    message(FATAL_ERROR "Core Vt requested version-specific Python")
endif()
get_target_property(features vt INTERFACE_COMPILE_FEATURES)
if(NOT "cxx_std_17" IN_LIST features)
    message(FATAL_ERROR "Core Vt must supply its C++17 requirement")
endif()
get_target_property(definitions vt INTERFACE_COMPILE_DEFINITIONS)
if(definitions MATCHES "Py_LIMITED_API")
    message(FATAL_ERROR "The core Limited API definition leaks to bindings")
endif()

function(check_dependencies target)
    get_property(visited GLOBAL PROPERTY checkedTargets)
    if(target IN_LIST visited)
        return()
    endif()
    set_property(GLOBAL APPEND PROPERTY checkedTargets "${target}")
    get_target_property(dependencies "${target}" INTERFACE_LINK_LIBRARIES)
    foreach(dependency IN LISTS dependencies)
        if(dependency MATCHES "(^|::)(boost|python)(::|$)|(tf|vt)BoostPython"
                OR (WIN32 AND dependency STREQUAL "Python3::Python"))
            message(FATAL_ERROR "${target} exposes binding dependency ${dependency}")
        endif()
        if(TARGET "${dependency}")
            check_dependencies("${dependency}")
        endif()
    endforeach()
endfunction()
check_dependencies(vt)

if(WIN32)
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
endif()
file(STRINGS "${CORE_HEADERS}" headers)
set(headerSource "struct VtCoreHeaderVisitType {};\n")
foreach(header IN LISTS headers)
    if(header STREQUAL "addExtVisitValueType.h")
        string(APPEND headerSource
            "#define VT_ADD_EXT_VISIT_VALUE_TYPE VtCoreHeaderVisitType\n")
    endif()
    string(APPEND headerSource "#include \"pxr/base/vt/${header}\"\n")
endforeach()
string(APPEND headerSource "
#if defined(VT_EXPECT_FULL_API) && defined(Py_LIMITED_API)
#error The core must not impose the Limited API on its consumers.
#endif
")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/headers.cpp" "${headerSource}")
foreach(mode Full Limited)
    add_library(vtHeaders${mode} OBJECT "${CMAKE_CURRENT_BINARY_DIR}/headers.cpp")
    target_link_libraries(vtHeaders${mode} PRIVATE vt)
    if(mode STREQUAL "Limited")
        target_compile_definitions(vtHeaders${mode} PRIVATE Py_LIMITED_API=0x030A0000)
    else()
        target_compile_definitions(vtHeaders${mode} PRIVATE VT_EXPECT_FULL_API)
    endif()
endforeach()

file(STRINGS "${CORE_SOURCES}" sources)
add_library(vtSourcesLimited OBJECT ${sources})
target_link_libraries(vtSourcesLimited PRIVATE vt)
target_compile_definitions(vtSourcesLimited PRIVATE VT_EXPORTS
    Py_LIMITED_API=0x030A0000 MFB_PACKAGE_NAME=vt MFB_ALT_PACKAGE_NAME=vt MFB_PACKAGE_MODULE=Vt)
foreach(target vtHeadersFull vtHeadersLimited vtSourcesLimited)
    if(MSVC)
        target_compile_options(${target} PRIVATE /showIncludes)
    else()
        target_compile_options(${target} PRIVATE -H)
    endif()
endforeach()

add_executable(vtCoreConsumer consumer.cpp)
target_link_libraries(vtCoreConsumer PRIVATE vt)
target_compile_definitions(vtCoreConsumer PRIVATE Py_LIMITED_API=0x030A0000)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/consumer-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:vtCoreConsumer>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/core-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:vt>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/runtime-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE_DIR:TBB::tbb>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/binding-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE_NAME:python>")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/linker.txt" "${CMAKE_LINKER}")
