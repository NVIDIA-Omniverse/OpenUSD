cmake_minimum_required(VERSION 3.26)
project(VtConversionRegistry LANGUAGES CXX)
file(TO_CMAKE_PATH "${USD_PREFIX}" USD_PREFIX)
find_package(pxr CONFIG REQUIRED COMPONENTS vtBoostPython
    PATHS "${USD_PREFIX}" NO_DEFAULT_PATH)
get_target_property(bindingType vtBoostPython TYPE)
if(NOT bindingType STREQUAL "INTERFACE_LIBRARY")
    message(FATAL_ERROR "Vt binding helpers must not add a compiled library")
endif()
get_target_property(headers vtBoostPython HEADER_SET)
if(NOT "${USD_PREFIX}/include/pxr/base/vt/boost/valueFromPythonBoost.h" IN_LIST headers)
    message(FATAL_ERROR "The binding interface must own its adapter header")
endif()
if(WIN32)
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()

# Check the neutral header independently of value.h, which is migrated later.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/header.cpp"
    "#include \"pxr/base/vt/valueFromPython.h\"\n")
foreach(mode full limited)
    add_library(registryHeader_${mode} OBJECT "${CMAKE_CURRENT_BINARY_DIR}/header.cpp")
    target_link_libraries(registryHeader_${mode} PRIVATE tf)
    target_compile_features(registryHeader_${mode} PRIVATE cxx_std_17)
    if(mode STREQUAL "limited")
        target_compile_definitions(registryHeader_${mode} PRIVATE Py_LIMITED_API=0x030A0000)
    endif()
    if(MSVC)
        target_compile_options(registryHeader_${mode} PRIVATE /showIncludes)
    endif()
endforeach()

add_executable(registryContracts contracts.cpp)
target_compile_features(registryContracts PRIVATE cxx_std_17)
target_compile_definitions(registryContracts PRIVATE Py_LIMITED_API=0x030A0000)
if(TARGET usd_m)
    get_target_property(coreType usd_m TYPE)
    if(coreType STREQUAL "STATIC_LIBRARY")
        target_link_libraries(registryContracts PRIVATE "$<LINK_LIBRARY:WHOLE_ARCHIVE,usd_m>")
    else()
        target_link_libraries(registryContracts PRIVATE usd_m)
    endif()
else()
    target_link_libraries(registryContracts PRIVATE vt)
endif()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/contracts-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:registryContracts>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/runtime-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE_DIR:TBB::tbb>")

foreach(module vtRegistryRegister vtRegistryConsume)
    if(module STREQUAL "vtRegistryRegister")
        set(source register.cpp)
    else()
        set(source consume.cpp)
    endif()
    add_library(${module} MODULE ${source})
    target_link_libraries(${module} PRIVATE vtBoostPython)
    set_target_properties(${module} PROPERTIES PREFIX "")
    if(WIN32)
        set_target_properties(${module} PROPERTIES SUFFIX ".pyd")
    elseif(APPLE)
        target_link_options(${module} PRIVATE "LINKER:-undefined,dynamic_lookup")
    endif()
endforeach()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/module-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:vtRegistryConsume>")
