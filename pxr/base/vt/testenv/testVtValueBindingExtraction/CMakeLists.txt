cmake_minimum_required(VERSION 3.26)
project(VtValueBindingExtraction LANGUAGES CXX)
file(TO_CMAKE_PATH "${USD_PREFIX}" USD_PREFIX)
find_package(pxr CONFIG REQUIRED COMPONENTS vtBoostPython
    PATHS "${USD_PREFIX}" NO_DEFAULT_PATH)
if(WIN32)
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(TARGET usd_m)
    set(coreTarget usd_m)
    get_target_property(coreType usd_m TYPE)
    if(coreType STREQUAL "STATIC_LIBRARY")
        set(core "$<LINK_LIBRARY:WHOLE_ARCHIVE,usd_m>")
    else()
        set(core usd_m)
    endif()
else()
    set(core vt)
    set(coreTarget vt)
endif()

add_library(vtNativeTypes SHARED nativeTypes.cpp)
target_compile_definitions(vtNativeTypes PRIVATE VT_NATIVE_TYPES_EXPORTS)
target_compile_features(vtNativeTypes PUBLIC cxx_std_17)
target_link_libraries(vtNativeTypes PUBLIC ${core})
add_executable(valueContracts contracts.cpp)
target_link_libraries(valueContracts PRIVATE vtNativeTypes)
foreach(target vtNativeTypes valueContracts)
    target_compile_definitions(${target} PRIVATE Py_LIMITED_API=0x030A0000)
    if(MSVC)
        target_compile_options(${target} PRIVATE /showIncludes)
    endif()
endforeach()

file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/headers.cpp"
    "#include \"pxr/base/vt/valueCommon.h\"\n#include \"pxr/base/vt/valueRef.h\"\n#include \"pxr/base/vt/value.h\"\n")
foreach(mode full limited)
    add_library(valueHeader_${mode} OBJECT "${CMAKE_CURRENT_BINARY_DIR}/headers.cpp")
    target_link_libraries(valueHeader_${mode} PRIVATE ${core})
    target_compile_features(valueHeader_${mode} PRIVATE cxx_std_17)
    if(mode STREQUAL "limited")
        target_compile_definitions(valueHeader_${mode} PRIVATE Py_LIMITED_API=0x030A0000)
    endif()
    if(MSVC)
        target_compile_options(valueHeader_${mode} PRIVATE /showIncludes)
    endif()
endforeach()

# Compile the actual migrated implementation units against the Limited API.
# These objects are not linked: the installed SDK supplies their runtime code.
add_library(valueSourcesLimited OBJECT
    ../../value.cpp ../../valueRef.cpp ../../valueCommon.cpp
    ../../valueFromPython.cpp ../../valueToPython.cpp)
target_link_libraries(valueSourcesLimited PRIVATE ${core})
target_compile_features(valueSourcesLimited PRIVATE cxx_std_17)
target_compile_definitions(valueSourcesLimited PRIVATE
    VT_EXPORTS Py_LIMITED_API=0x030A0000
    MFB_PACKAGE_NAME=vt MFB_PACKAGE_MODULE=Vt MFB_ALT_PACKAGE_NAME=vt)
if(MSVC)
    target_compile_options(valueSourcesLimited PRIVATE /showIncludes)
endif()

add_library(missingAdapter OBJECT EXCLUDE_FROM_ALL missingAdapter.cpp)
target_link_libraries(missingAdapter PRIVATE vtBoostPython)
add_library(bindingFirst OBJECT bindingFirst.cpp)
target_link_libraries(bindingFirst PRIVATE vtBoostPython)
foreach(module vtValueRegister vtValueConsume)
    if(module STREQUAL "vtValueRegister")
        set(source register.cpp)
    else()
        set(source consume.cpp)
    endif()
    add_library(${module} MODULE ${source})
    target_link_libraries(${module} PRIVATE vtBoostPython vtNativeTypes)
    set_target_properties(${module} PROPERTIES PREFIX "")
    if(WIN32)
        set_target_properties(${module} PROPERTIES SUFFIX ".pyd")
    elseif(APPLE)
        target_link_options(${module} PRIVATE "LINKER:-undefined,dynamic_lookup")
    endif()
endforeach()
get_target_property(bindingHeaders vtBoostPython HEADER_SET)
foreach(header valueBoost.h valueToPythonBoost.h)
    if(NOT "${USD_PREFIX}/include/pxr/base/vt/boost/${header}" IN_LIST bindingHeaders)
        message(FATAL_ERROR "Binding target does not own ${header}")
    endif()
endforeach()
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/contracts-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:valueContracts>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/native-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:vtNativeTypes>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/core-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:${coreTarget}>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/module-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:vtValueConsume>")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/runtime-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE_DIR:TBB::tbb>")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/linker.txt" "${CMAKE_LINKER}")
get_target_property(bindingRuntimeType python TYPE)
if(bindingRuntimeType STREQUAL "SHARED_LIBRARY")
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/binding-$<CONFIG>.txt"
        CONTENT "$<TARGET_FILE_NAME:python>")
else()
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/binding-$<CONFIG>.txt"
        CONTENT "")
endif()
